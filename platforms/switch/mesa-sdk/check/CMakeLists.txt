cmake_minimum_required(VERSION 3.25)
project(dusklight_nvk_sdk_check LANGUAGES C CXX)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_LIST_DIR}/../../cmake")
find_package(NXVK REQUIRED)
if(NOT TARGET Vulkan::Vulkan)
  message(FATAL_ERROR "SDK mode must expose the upstream Vulkan::Vulkan contract")
endif()
add_executable(nvk_sdk_check main.cpp ../../source/switch_nvk_sdk_dispatch.cpp
  ../../source/switch_posix_shim.cpp)
target_compile_definitions(nvk_sdk_check PRIVATE DUSK_SWITCH_NVK_UPSTREAM_SDK=1)
target_include_directories(nvk_sdk_check PRIVATE ../../include)
target_compile_features(nvk_sdk_check PRIVATE cxx_std_20)
target_link_libraries(nvk_sdk_check PRIVATE NXVK::nvk)
target_link_options(nvk_sdk_check PRIVATE "-Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/nvk_sdk_check.map")
target_link_options(nvk_sdk_check PRIVATE "-Wl,--wrap=vk_icdGetInstanceProcAddr")
target_link_options(nvk_sdk_check PRIVATE "-Wl,--wrap=posix_memalign")
