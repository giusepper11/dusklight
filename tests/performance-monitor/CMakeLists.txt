cmake_minimum_required(VERSION 3.25)
project(dusklight_performance_monitor_host_tests LANGUAGES C CXX)
if(CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "Host tests cannot run as Switch executables")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(AURORA "${ROOT}/extern/aurora")
enable_testing()
find_package(Threads REQUIRED)
add_library(performance_capture STATIC
    "${ROOT}/src/dusk/performance_capture.cpp" "${ROOT}/src/dusk/performance_csv.cpp")
target_include_directories(performance_capture PUBLIC "${ROOT}/src")
target_compile_options(performance_capture PRIVATE -Wall -Wextra -Werror)
foreach(name IN ITEMS performance_monitor performance_csv performance_deadline)
    add_executable(${name}_test "${ROOT}/tests/${name}_test.cpp")
    target_link_libraries(${name}_test PRIVATE performance_capture Threads::Threads)
    target_compile_options(${name}_test PRIVATE -Wall -Wextra -Werror)
endforeach()
add_test(NAME performance_measurement COMMAND performance_monitor_test)
add_test(NAME performance_csv COMMAND performance_csv_test "${CMAKE_CURRENT_BINARY_DIR}/csv-fixtures")
add_test(NAME performance_deadline_regressions COMMAND performance_deadline_test)
add_executable(present_observer_test "${ROOT}/tests/present_observer_test.cpp")
target_include_directories(present_observer_test PRIVATE "${AURORA}/include" "${AURORA}/lib")
target_link_libraries(present_observer_test PRIVATE Threads::Threads)
target_compile_options(present_observer_test PRIVATE -Wall -Wextra -Werror)
add_test(NAME aurora_present_observer COMMAND present_observer_test)

# Run the ACTUAL pinned upstream CPU/thread tests, without Dawn or a GPU.
foreach(dependency IN ITEMS SDL_SOURCE GTEST_SOURCE TRACY_SOURCE)
    if(NOT IS_DIRECTORY "${${dependency}}")
        message(FATAL_ERROR "Set ${dependency} to a verified local dependency source")
    endif()
endforeach()
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_UNIX_CONSOLE_BUILD ON CACHE BOOL "" FORCE)
foreach(feature IN ITEMS X11 WAYLAND AUDIO VIDEO JOYSTICK HAPTIC HIDAPI SENSOR CAMERA DIALOG)
    set(SDL_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${SDL_SOURCE}" sdl EXCLUDE_FROM_ALL)
add_subdirectory("${GTEST_SOURCE}" googletest EXCLUDE_FROM_ALL)
include(GoogleTest)
add_executable(aurora_io_tests "${AURORA}/tests/io_test.cpp" "${AURORA}/lib/card/FileIO.cpp" "${AURORA}/lib/io.cpp")
target_include_directories(aurora_io_tests PRIVATE "${AURORA}/lib" "${AURORA}/include")
target_link_libraries(aurora_io_tests PRIVATE GTest::gtest_main SDL3::SDL3-static)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/io-test-tmp")
# Upstream fixtures use a per-process counter starting at zero. Isolate their
# owned temporary directory and serialize the individual CTest invocations.
gtest_discover_tests(aurora_io_tests PROPERTIES RUN_SERIAL TRUE
    ENVIRONMENT "TMPDIR=${CMAKE_CURRENT_BINARY_DIR}/io-test-tmp")
add_executable(aurora_time_tests "${AURORA}/tests/time_test.cpp" "${AURORA}/lib/time.cpp")
target_include_directories(aurora_time_tests PRIVATE "${AURORA}/include" "${AURORA}/lib")
target_link_libraries(aurora_time_tests PRIVATE GTest::gtest_main)
gtest_discover_tests(aurora_time_tests)
add_executable(aurora_render_worker_tests "${AURORA}/tests/render_worker_test.cpp"
    "${AURORA}/tests/thread_test.cpp" "${AURORA}/lib/gfx/render_worker.cpp" "${AURORA}/lib/thread.cpp")
target_include_directories(aurora_render_worker_tests PRIVATE "${AURORA}/include" "${AURORA}/lib" "${TRACY_SOURCE}/public")
target_link_libraries(aurora_render_worker_tests PRIVATE GTest::gtest_main SDL3::SDL3-static Threads::Threads)
gtest_discover_tests(aurora_render_worker_tests)
